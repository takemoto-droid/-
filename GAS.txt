function ss_() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

function ensureSheet_(name) {
  var ss = ss_();
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  var headers = HEADERS[name];
  var first = sheet.getRange(1, 1, 1, Math.max(headers.length, 1)).getValues()[0];
  var needsHeader = headers.some(function (h, i) { return first[i] !== h; });
  if (needsHeader) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function sheet_(name) {
  var sheet = ss_().getSheetByName(name);
  if (!sheet) sheet = ensureSheet_(name);
  return sheet;
}

/** シートの全行をヘッダー名をキーにしたオブジェクト配列として読む */
function readSheetAsObjects_(sheet) {
  var lastRow = sheet.getLastRow();
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  if (lastRow < 1) return { headers: [], rows: [] };
  var values = sheet.getRange(1, 1, lastRow, lastCol).getValues();
  var headers = values[0].map(function (h) { return String(h).trim(); });
  var rows = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    var obj = {};
    var empty = true;
    for (var c = 0; c < headers.length; c++) {
      if (!headers[c]) continue;
      obj[headers[c]] = row[c];
      if (row[c] !== '' && row[c] !== null && row[c] !== undefined) empty = false;
    }
    if (!empty) {
      obj.__row = i + 1; // 1-based row number in sheet
      rows.push(obj);
    }
  }
  return { headers: headers, rows: rows };
}

function objectToRow_(headers, obj) {
  return headers.map(function (h) {
    var v = obj[h];
    return (v === undefined || v === null) ? '' : v;
  });
}

function appendObjectRow_(sheet, headers, obj) {
  sheet.appendRow(objectToRow_(headers, obj));
}

function appendObjectRows_(sheet, headers, objs) {
  if (!objs.length) return;
  var rows = objs.map(function (o) { return objectToRow_(headers, o); });
  var startRow = sheet.getLastRow() + 1;
  sheet.getRange(startRow, 1, rows.length, headers.length).setValues(rows);
}

function writeObjectRow_(sheet, headers, rowIndex1based, obj) {
  sheet.getRange(rowIndex1based, 1, 1, headers.length).setValues([objectToRow_(headers, obj)]);
}

/** 条件に合う行を下から上へ削除する（インデックスずれ防止） */
function deleteRowsWhere_(sheet, rows, predicate) {
  var toDelete = rows.filter(predicate).map(function (r) { return r.__row; }).sort(function (a, b) { return b - a; });
  toDelete.forEach(function (r) { sheet.deleteRow(r); });
  return toDelete.length;
}

function withLock_(fn) {
  var lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    return fn();
  } finally {
    lock.releaseLock();
  }
}

// ---- CacheService ラッパ（put失敗は握りつぶす。100KB/key超はスキップ） ----

var CACHE_MAX_BYTES = 100000; // CacheService の 1キー上限（約100KB）

function cache_() {
  return CacheService.getScriptCache();
}

function cacheGet_(key) {
  try {
    return cache_().get(key);
  } catch (e) {
    return null;
  }
}

function cacheGetJson_(key) {
  var raw = cacheGet_(key);
  if (raw === null || raw === undefined) return null;
  try {
    return JSON.parse(raw);
  } catch (e) {
    return null;
  }
}

/** put失敗（サイズ超過・サービスエラー）は無視する。valueは文字列想定 */
function cachePut_(key, value, ttlSec) {
  try {
    if (value.length > CACHE_MAX_BYTES) return; // サイズ超過はキャッシュしない
    cache_().put(key, value, ttlSec);
  } catch (e) {
    // put失敗は無視（フォールバックで都度再計算する）
  }
}

function cachePutJson_(key, obj, ttlSec) {
  try {
    cachePut_(key, JSON.stringify(obj), ttlSec);
  } catch (e) {
    // JSON化失敗も無視
  }
}

function cacheRemove_(keys) {
  try {
    cache_().removeAll(keys);
  } catch (e) {
    // 削除失敗は無視
  }
}

/** answers を (student_id, target_date, question_id) の最後の行のみに絞る */
function dedupeAnswers_(rows) {
  var byKey = {};
  var order = [];
  rows.forEach(function (r) {
    var key = String(r.student_id).trim() + '|' + normDateStr_(r.target_date) + '|' + String(r.question_id).trim();
    if (!(key in byKey)) order.push(key);
    byKey[key] = r; // 後勝ち（最後の行）
  });
  return order.map(function (k) { return byKey[k]; });
}

/** submissions を (student_id, target_date) の最初の行のみに絞る */
function dedupeSubmissions_(rows) {
  var byKey = {};
  var order = [];
  rows.forEach(function (r) {
    var key = String(r.student_id).trim() + '|' + normDateStr_(r.target_date);
    if (!(key in byKey)) { byKey[key] = r; order.push(key); } // 先勝ち（最初の行）
  });
  return order.map(function (k) { return byKey[k]; });
}

    // ログ失敗は握りつぶす（本処理を止めない）
  }
}

function fmtDate_(d) {
  return Utilities.formatDate(d, TZ, 'yyyy-MM-dd');
}

/** セルの値が Date でも文字列でも 'yyyy-MM-dd' 文字列に正規化する */
function normDateStr_(v) {
  if (v instanceof Date) return fmtDate_(v);
  return String(v || '').trim();
}

function addDaysStr_(dateStr, delta) {
  var p = dateStr.split('-').map(Number);
  var d = new Date(p[0], p[1] - 1, p[2]);
  d.setDate(d.getDate() + delta);
  return fmtDate_(d);
}

function dateStrToDate_(dateStr) {
  var p = dateStr.split('-').map(Number);
  return new Date(p[0], p[1] - 1, p[2]);
}

function weekdayJp_(dateStr) {
  return WEEKDAY_JP[dateStrToDate_(dateStr).getDay()];
}

/** 月曜始まりの週の月曜日の日付文字列を返す */
function mondayOfWeek_(dateStr) {
  var d = dateStrToDate_(dateStr);
  var dow = d.getDay(); // 0=日 1=月...
  var backToMonday = (dow === 0) ? 6 : (dow - 1);
  d.setDate(d.getDate() - backToMonday);
  return fmtDate_(d);
}

function monthOf_(dateStr) {
  return dateStr.slice(0, 7); // yyyy-MM
}

function getSettingsMap_() {
  if (MEMO_SETTINGS) return MEMO_SETTINGS;
  var cached = cacheGetJson_(CACHE_KEY_SETTINGS);
  if (cached) { MEMO_SETTINGS = cached; return cached; }

  var sheet = sheet_(SHEETS.SETTINGS);
  var data = readSheetAsObjects_(sheet).rows;
  var map = {};
  data.forEach(function (r) { map[String(r.key).trim()] = r.value; });
  DEFAULT_SETTINGS.forEach(function (kv) {
    if (!(kv[0] in map)) map[kv[0]] = kv[1];
  });
  var out = {
    join_bonus: Number(map.join_bonus),
    pt_1: Number(map.pt_1),
    pt_2: Number(map.pt_2),
    pt_3: Number(map.pt_3),
    makeup_days: Number(map.makeup_days),
    date_override: normDateStr_(map.date_override),
    demo_mode: isTrue_(map.demo_mode)
  };
  cachePutJson_(CACHE_KEY_SETTINGS, out, CACHE_TTL_SETTINGS);
  MEMO_SETTINGS = out;
  return out;
}

function resolveCourse_(rosterRow, courseMap) {
  if (rosterRow.course && COURSE_VALUES.indexOf(rosterRow.course) !== -1) return rosterRow.course;

function getQuestionsMap_() {
  var sheet = sheet_(SHEETS.QUESTIONS);
  var rows = readSheetAsObjects_(sheet).rows;
  var map = {};
  rows.forEach(function (r) {
    var qid = String(r.question_id || '').trim();
    if (!qid) return;
    map[qid] = {
      question_id: qid,
      publish_date: normDateStr_(r.publish_date),
      seq: Number(r.seq),
      genre: String(r.genre || '').trim(),
      difficulty: Number(r.difficulty),
      type: String(r.type || '').trim(),
      passage: String(r.passage || ''),
      question_text: String(r.question_text || ''),
      choices: [String(r.choice1 || ''), String(r.choice2 || ''), String(r.choice3 || ''), String(r.choice4 || '')],
      answer: Number(r.answer),
      explanation: String(r.explanation || ''),
      time_limit_sec: Number(r.time_limit_sec) || 60
    };
  });
  return map;
}

function questionsForDate_(questionsMap, dateStr) {
  var list = [];
  for (var id in questionsMap) {
    if (questionsMap[id].publish_date === dateStr) list.push(questionsMap[id]);
  }
  list.sort(function (a, b) { return a.seq - b.seq; });
  return list;
}

/** 特定日の問題配列だけが必要な箇所用の日付スライスキャッシュ（q:{date}, TTL6h）。
 *  ミス時のみ全問マップを構築してその日分を切り出す。put失敗（サイズ超）はフォールバック。 */
function getQuestionsForDate_(dateStr) {
  var cached = cacheGetJson_(cacheKeyQ_(dateStr));
  if (cached) return cached;
  var list = questionsForDate_(getQuestionsMap_(), dateStr);
  cachePutJson_(cacheKeyQ_(dateStr), list, CACHE_TTL_Q);
  return list;
}

/** 問題キャッシュ（日付スライス）を全消去する。CSV取込後に呼ぶ。
 *  日付キーは列挙できないため、questions シートの publish_date を走査して該当キーを削除。 */
function invalidateQuestionsCache_() {
  try {
    var rows = readSheetAsObjects_(sheet_(SHEETS.QUESTIONS)).rows;
    var dateSet = {};
    rows.forEach(function (r) { dateSet[normDateStr_(r.publish_date)] = true; });
    var keys = Object.keys(dateSet).filter(function (d) { return d; }).map(cacheKeyQ_);
    if (keys.length) cacheRemove_(keys);
  } catch (e) {
    // 走査失敗時は無視（TTLで自然失効）
  }
}

// 決定的シャッフル（studentId+date+questionId をシードに使う）
function seededPermutation4_(seedStr) {
  var h = 1779033703 ^ seedStr.length;
  for (var i = 0; i < seedStr.length; i++) {
    h = Math.imul(h ^ seedStr.charCodeAt(i), 3432918353);
    h = (h << 13) | (h >>> 19);
  }
  function next() {
    h = Math.imul(h ^ (h >>> 16), 2246822519);
    h = Math.imul(h ^ (h >>> 13), 3266489917);
    h ^= h >>> 16;
    return (h >>> 0) / 4294967296;
  }
  var arr = [0, 1, 2, 3];
  for (var i2 = 3; i2 > 0; i2--) {
    var j = Math.floor(next() * (i2 + 1));
    var tmp = arr[i2]; arr[i2] = arr[j]; arr[j] = tmp;
  }
  return arr; // arr[表示位置] = 元のchoice index(0-based)
}

function getBootstrap(studentId) {
  try {
    studentId = String(studentId || '').trim().toUpperCase();
    if (!studentId) return { ok: false, error: 'IDを入力してください。' };

    var roster = getRosterMap_();
    var row = roster[studentId];
    if (!row || !row.active) {

    }

    var settings = getSettingsMap_();
    var courseMap = getCourseMap_();

        hasQuestions: hasQ
      });
    }

    return {
      ok: true,
      student: {
        id: studentId,
        displayName: displayNameOf_(row),
        nickname: row.nickname,
        class: row.class,
        course: course,
        role: row.role
      },

        join_bonus: settings.join_bonus, makeup_days: settings.makeup_days,
        demo_mode: settings.demo_mode
      },
      alreadyDone: !!doneDates[today]
    };
  } catch (err) {
    return { ok: false, error: '通信エラー: ' + err };
  }
}

var NICKNAME_RE = /^[0-9A-Za-zぁ-ゖァ-ヺー一-鿿ｦ-ﾝ]{1,10}$/;

function setNickname(studentId, nickname) {
  return withLock_(function () {
    try {
      studentId = String(studentId || '').trim().toUpperCase();
      nickname = String(nickname || '').trim();
      var roster = getRosterMap_();
      var row = roster[studentId];
      if (!row) return { ok: false, error: 'IDが見つかりません。' };

      if (nickname) {
        if (nickname.length > 10 || !NICKNAME_RE.test(nickname)) {
          return { ok: false, error: 'ニックネームは10文字以内・ひらがな/カタカナ/漢字/英数字のみ使えます。' };
        }
        for (var id in roster) {
          if (id === studentId) continue;
          if (id === nickname || roster[id].nickname === nickname) {
            return { ok: false, error: 'そのニックネームは使用できません（他の人のIDと重複します）。' };
          }
        }
      }

      var sheet = sheet_(SHEETS.ROSTER);
      var oldNickname = row.nickname;
      var nicknameCol = HEADERS[SHEETS.ROSTER].indexOf('nickname') + 1;
      sheet.getRange(row.__row, nicknameCol, 1, 1).setValue(nickname);
      invalidateRosterCache_(); // roster変更のためキャッシュ削除
      logEvent_('nickname_change', studentId, oldNickname + ' -> ' + nickname);

      return { ok: true, displayName: nickname || studentId };
    } catch (err) {
      return { ok: false, error: '通信エラー: ' + err };
    }
  });
}

function getQuiz(studentId, targetDate) {
  try {
    studentId = String(studentId || '').trim().toUpperCase();
    targetDate = normDateStr_(targetDate);
    var roster = getRosterMap_();
    var row = roster[studentId];
    if (!row || !row.active) return { ok: false, error: 'IDが見つかりません。' };

    var settings = getSettingsMap_();
    var today = todayStr_(settings);
    var isMakeup = targetDate !== today;

    if (!isMakeup && !settings.demo_mode) {
      var subs = dedupeSubmissions_(readSheetAsObjects_(sheet_(SHEETS.SUBMISSIONS)).rows);
      var already = subs.some(function (s) {
        return String(s.student_id).trim() === studentId && normDateStr_(s.target_date) === targetDate;
      });
      if (already) return { ok: false, error: '本日分は回答済みです。' };
    }

    var qs = getQuestionsForDate_(targetDate);
    if (!qs.length) return { ok: false, error: 'この日の問題はまだ配信されていません。' };

    var out = qs.map(function (q) {
      var perm = seededPermutation4_(studentId + '|' + targetDate + '|' + q.question_id);
      var shuffledChoices = perm.map(function (origIdx) { return q.choices[origIdx]; });
      return {
        question_id: q.question_id,
        seq: q.seq,
        genre: q.genre,
        difficulty: q.difficulty,
        type: q.type,
        passage: q.passage,
        question_text: q.question_text,
        choices: shuffledChoices,
        dispToOrig: perm, // dispToOrig[表示位置(0始まり)] = 元のchoice index(0始まり)
        time_limit_sec: q.time_limit_sec
      };
    });

    return { ok: true, questions: out, isMakeup: isMakeup, targetDate: targetDate };
  } catch (err) {
    return { ok: false, error: '通信エラー: ' + err };
  }
}

function submitAnswer(studentId, targetDate, questionId, selectedOrigIndex, elapsedSec) {
  try {
    studentId = String(studentId || '').trim().toUpperCase();
    targetDate = normDateStr_(targetDate);
    questionId = String(questionId || '').trim();
    var q = getQuestionsMap_()[questionId];
    if (!q) return { ok: false, error: '問題が見つかりません。' };

    var settings = getSettingsMap_();
    var today = todayStr_(settings);
    var isMakeup = targetDate !== today;

    // demo_mode のみ従来パス（ロック＋シート読み＋既存行削除）を維持
    if (settings.demo_mode) {
      return withLock_(function () {
        var sheet = sheet_(SHEETS.ANSWERS);
        var rows = readSheetAsObjects_(sheet).rows;
        deleteRowsWhere_(sheet, rows, function (r) {
          return String(r.student_id).trim() === studentId &&
            normDateStr_(r.target_date) === targetDate &&
            String(r.question_id).trim() === questionId;
        });
        return recordAnswer_(sheet, studentId, targetDate, questionId, q, settings, selectedOrigIndex, elapsedSec, isMakeup);
      });
    }

    // 通常パス（ロックなし）: 重複チェックは Cache（ans:{sid}:{date}:{qid}）で行う
    var ansKey = cacheKeyAns_(studentId, targetDate, questionId);
    var cached = cacheGetJson_(ansKey);
    if (cached) return cached; // 二重送信は直前の結果を返す

    var res = recordAnswer_(sheet_(SHEETS.ANSWERS), studentId, targetDate, questionId, q, settings, selectedOrigIndex, elapsedSec, isMakeup);
    if (res.ok) cachePutJson_(ansKey, res, CACHE_TTL_ANS); // 書込成功直後にset
    return res;
  } catch (err) {
    return { ok: false, error: '通信エラー: ' + err };
  }
}

/** answers への1行追記と判定結果の生成（stats更新はしない=P1で日次化） */
function recordAnswer_(sheet, studentId, targetDate, questionId, q, settings, selectedOrigIndex, elapsedSec, isMakeup) {
  var origIdx1based = Number(selectedOrigIndex); // 0=時間切れ, 1-4=選択
  var isCorrect = origIdx1based > 0 && origIdx1based === q.answer;
  var point = isCorrect ? ptForDifficulty_(settings, q.difficulty) : 0;

  appendObjectRow_(sheet, HEADERS[SHEETS.ANSWERS], {
    timestamp: new Date(),
    student_id: studentId,
    target_date: targetDate,
    question_id: questionId,
    selected: origIdx1based,
    is_correct: isCorrect,
    elapsed_sec: Number(elapsedSec) || 0,
    is_makeup: isMakeup
  });

  return { ok: true, correct: isCorrect, answer: q.answer, explanation: q.explanation, point: point };
}

/** P1で日次化: 呼び出しは撤去済み。dailyAggregate内で全期間から再計算する（互換のため関数は温存） */
function updateStatsQuestion_(questionId, isCorrect) {
  var sheet = sheet_(SHEETS.STATS_QUESTION);
  var data = readSheetAsObjects_(sheet);
  var found = data.rows.filter(function (r) { return String(r.question_id).trim() === questionId; })[0];
  if (found) {
    var attempts = Number(found.attempts || 0) + 1;
    var correct = Number(found.correct || 0) + (isCorrect ? 1 : 0);
    writeObjectRow_(sheet, HEADERS[SHEETS.STATS_QUESTION], found.__row, {
      question_id: questionId, attempts: attempts, correct: correct, correct_rate: correct / attempts
    });
  } else {
    appendObjectRow_(sheet, HEADERS[SHEETS.STATS_QUESTION], {
      question_id: questionId, attempts: 1, correct: isCorrect ? 1 : 0, correct_rate: isCorrect ? 1 : 0
    });
  }
}

function finishQuiz(studentId, targetDate) {
  try {
    studentId = String(studentId || '').trim().toUpperCase();
    targetDate = normDateStr_(targetDate);
    var roster = getRosterMap_();
    var row = roster[studentId];
    if (!row) return { ok: false, error: 'IDが見つかりません。' };

    var settings = getSettingsMap_();
    var today = todayStr_(settings);
    var isMakeup = targetDate !== today;
    var doneKey = cacheKeyDone_(studentId, targetDate);
    var subSheet = sheet_(SHEETS.SUBMISSIONS);

    if (settings.demo_mode) {
      // demo_modeのみロック内で既存提出を削除（提出のやり直しを許可）
      withLock_(function () {
        var subData = readSheetAsObjects_(subSheet);
        deleteRowsWhere_(subSheet, subData.rows, function (s) {
          return String(s.student_id).trim() === studentId && normDateStr_(s.target_date) === targetDate;
        });
      });
    } else {
      // 提出済みチェック: Cache先行判定（ヒット→即「提出済み」）
      if (cacheGet_(doneKey)) return { ok: false, error: '本日分は既に提出済みです。' };
      var existing = dedupeSubmissions_(readSheetAsObjects_(subSheet).rows).filter(function (s) {
        return String(s.student_id).trim() === studentId && normDateStr_(s.target_date) === targetDate;
      });
      if (existing.length) {
        cachePut_(doneKey, '1', CACHE_TTL_DONE);
        return { ok: false, error: '本日分は既に提出済みです。' };
      }
    }

    var questionsMap = getQuestionsMap_();
    var answerRows = dedupeAnswers_(readSheetAsObjects_(sheet_(SHEETS.ANSWERS)).rows).filter(function (a) {
      return String(a.student_id).trim() === studentId && normDateStr_(a.target_date) === targetDate;
    });

    var totalPoint = 0;
    var totalElapsed = 0;
    var genreStats = {};
    answerRows.forEach(function (a) {
      var q = questionsMap[String(a.question_id).trim()];
      if (!q) return;
      var ok = isTrue_(a.is_correct);
      if (ok) totalPoint += ptForDifficulty_(settings, q.difficulty);
      totalElapsed += Number(a.elapsed_sec) || 0;
      var g = q.genre || '未分類';
      genreStats[g] = genreStats[g] || { ok: 0, all: 0 };
      genreStats[g].all++;
      if (ok) genreStats[g].ok++;
    });

    var joinBonus = settings.join_bonus;
    var totalScore = totalPoint + joinBonus;

    // append はロック外（append-only はアトミック）
    appendObjectRow_(subSheet, HEADERS[SHEETS.SUBMISSIONS], {
      student_id: studentId,
      target_date: targetDate,
      total_score: totalScore,
      total_elapsed_sec: totalElapsed,
      is_makeup: isMakeup,
      finished_at: new Date()
    });
    if (!settings.demo_mode) cachePut_(doneKey, '1', CACHE_TTL_DONE); // append後にset

    var out = { ok: true, totalScore: totalScore, joinBonus: joinBonus, genreStats: genreStats, isMakeup: isMakeup };

    // P7: 通常完了時はランキング結果を同梱し1往復化（isMakeup時は不要）
    if (!isMakeup) {
      var results = getResults(studentId, targetDate);
      if (results && results.ok) out.results = results;
    }
    return out;
  } catch (err) {
    return { ok: false, error: '通信エラー: ' + err };
  }
}

// ---- ランキング計算（getResults / dailyAggregate 共通ロジック） ----

function computeDailyRanking_(dateStr) {
  var roster = getRosterMap_();
  var courseMap = getCourseMap_();
  var subs = dedupeSubmissions_(readSheetAsObjects_(sheet_(SHEETS.SUBMISSIONS)).rows).filter(function (s) {
    return normDateStr_(s.target_date) === dateStr && !isTrue_(s.is_makeup);
  });

  var students = [];
  var teachers = [];

  subs.forEach(function (s) {
    var sid = String(s.student_id).trim();
    var r = roster[sid];
    if (!r) return;
    var entry = {
      student_id: sid,
      display_name: displayNameOf_(r),
      rank_name: rankingNameOf_(r),
      class: r.class,
      course: resolveCourse_(r, courseMap),
      role: r.role,
      homeroom_class: r.homeroom_class,
      score: Number(s.total_score) || 0,
      elapsed_sec: Number(s.total_elapsed_sec) || 0
    };
    if (r.role === 'teacher') teachers.push(entry);
    else students.push(entry);
  });

  function sortFn(a, b) {
    if (b.score !== a.score) return b.score - a.score;
    return a.elapsed_sec - b.elapsed_sec;
  }
  students.sort(sortFn);

  var n_grade = students.length;
  students.forEach(function (e, i) { e.rank_grade = i + 1; e.n_grade = n_grade; });

  ['class', 'course'].forEach(function (dim) {
    var groups = {};
    students.forEach(function (e) {
      var key = e[dim];
      if (!key) return;
      groups[key] = groups[key] || [];
      groups[key].push(e);
    });
    Object.keys(groups).forEach(function (key) {
      var arr = groups[key].slice().sort(sortFn);
      arr.forEach(function (e, i) {
        e['rank_' + dim] = i + 1;
        e['n_' + dim] = arr.length;
      });
    });
  });

  // クラス/コース集計（平均点・参加率）
  var rosterAll = Object.keys(roster).map(function (k) { return roster[k]; }).filter(function (r) { return r.role !== 'teacher' && r.active; });
  var classAgg = aggregateUnit_(students, rosterAll, 'class');
  var courseAgg = aggregateUnit_(students, rosterAll, 'course');

  return { students: students, teachers: teachers, classAgg: classAgg, courseAgg: courseAgg, n_grade: n_grade };
}

function aggregateUnit_(students, rosterAll, dim) {
  var totalByUnit = {};
  rosterAll.forEach(function (r) {
    var key = r[dim];
    if (!key) return;
    totalByUnit[key] = (totalByUnit[key] || 0) + 1;
  });
  var sumByUnit = {}, cntByUnit = {};
  students.forEach(function (e) {
    var key = e[dim];
    if (!key) return;
    sumByUnit[key] = (sumByUnit[key] || 0) + e.score;
    cntByUnit[key] = (cntByUnit[key] || 0) + 1;
  });
  var out = [];
  Object.keys(totalByUnit).forEach(function (key) {
    var n = cntByUnit[key] || 0;
    var total = totalByUnit[key];
    out.push({
      unit: key,
      avg_score: n ? (sumByUnit[key] / n) : 0,
      join_rate: total ? (n / total) : 0,
      n_participants: n,
      n_total: total
    });
  });
  out.sort(function (a, b) { return b.avg_score - a.avg_score; });
  return out;
}

/** 上位N名を取り、N位のscoreと同点の後続者も含めて返す（タイ含み方式）。
 *  list はスコア降順ソート済み想定。scoreKey省略時は 'score' を使う。 */
function topWithTies_(list, n, scoreKey) {
  scoreKey = scoreKey || 'score';
  if (!list.length) return [];
  if (list.length <= n) return list.slice();
  var cutoffScore = list[n - 1][scoreKey];
  var out = list.slice(0, n);
  for (var i = n; i < list.length; i++) {
    if (list[i][scoreKey] === cutoffScore) out.push(list[i]);
    else break;
  }
  return out;
}

function getResults(studentId, dateStr) {
  try {
    studentId = String(studentId || '').trim().toUpperCase();
    dateStr = normDateStr_(dateStr) || addDaysStr_(todayStr_(), -1);
    var agg = computeDailyRanking_(dateStr);

    var me = agg.students.filter(function (e) { return e.student_id === studentId; })[0];
    var myRanks = null;
    if (me) {
      myRanks = {
        class: { rank: me.rank_class || null, n: me.n_class || null },
        course: { rank: me.rank_course || null, n: me.n_course || null },
        grade: { rank: me.rank_grade, n: me.n_grade }
      };
    }

    // 学年TOP10（タイ含み）
    var gradeTop = topWithTies_(agg.students, 10).map(function (e) {
      return { rank: e.rank_grade, name: e.rank_name, score: e.score, isMe: e.student_id === studentId };
    });

    // 教員は参考行として学年TOP表示の下に追加提示（画面側で表示）
    var teacherRefs = agg.teachers.map(function (t) {
      return { name: t.rank_name + '（' + (t.role === 'teacher' ? '教員・参考' : '') + '）', score: t.score, class: t.class, course: t.course, homeroom_class: t.homeroom_class };
    });

    // 本人所属コースのみのTOP10（タイ含み）。コース一致の教員は全員 isRef で付与（現行仕様踏襲）
    var myCourseTop = [];
    if (me) {
      var courseList = agg.students.filter(function (e) { return e.course === me.course; })
        .sort(function (a, b) { return a.rank_course - b.rank_course; });
      myCourseTop = topWithTies_(courseList, 10, 'score').map(function (e) {
        return { rank: e.rank_course, name: e.rank_name, score: e.score, isMe: e.student_id === studentId };
      });
      var courseRefs = agg.teachers.filter(function (t) { return t.course === me.course; })
        .map(function (t) { return { name: t.rank_name, score: t.score, isRef: true }; });
      myCourseTop = myCourseTop.concat(courseRefs);
    }

    // 本人所属組のみのTOP10（タイ含み）。担任（homeroom_class一致）のみ isRef で付与
    var myClassTop = [];
    if (me) {
      var classList = agg.students.filter(function (e) { return e.class === me.class; })
        .sort(function (a, b) { return a.rank_class - b.rank_class; });
      myClassTop = topWithTies_(classList, 10, 'score').map(function (e) {
        return { rank: e.rank_class, name: e.rank_name, score: e.score, isMe: e.student_id === studentId };
      });
      var classRefs = agg.teachers.filter(function (t) { return t.homeroom_class === me.class; })
        .map(function (t) { return { name: t.rank_name, score: t.score, isRef: true }; });
      myClassTop = myClassTop.concat(classRefs);
    }

    // 正答率が低い問題（当日配信分）: stats_question参照をやめ、当日answersから直接算出
    var qsOfDay = getQuestionsForDate_(dateStr);
    var dayAnswers = dedupeAnswers_(readSheetAsObjects_(sheet_(SHEETS.ANSWERS)).rows).filter(function (a) {
      return normDateStr_(a.target_date) === dateStr;
    });
    var dayStat = {};
    dayAnswers.forEach(function (a) {
      var qid = String(a.question_id).trim();
      dayStat[qid] = dayStat[qid] || { attempts: 0, correct: 0 };
      dayStat[qid].attempts++;
      if (isTrue_(a.is_correct)) dayStat[qid].correct++;
    });
    var lowCorrectQuestions = qsOfDay
      .map(function (q) {
        var st = dayStat[q.question_id];
        var rate = st && st.attempts > 0 ? st.correct / st.attempts : null;
        return { question_id: q.question_id, question_text: q.question_text, explanation: q.explanation, correctRate: rate };
      })
      .filter(function (x) { return x.correctRate !== null; })
      .sort(function (a, b) { return a.correctRate - b.correctRate; })
      .slice(0, 3);

    return {
      ok: true,
      date: dateStr,
      gradeTop: gradeTop,
      teacherRefs: teacherRefs,
      myRanks: myRanks,
      myCourseTop: myCourseTop,
      myClassTop: myClassTop,
      lowCorrectQuestions: lowCorrectQuestions
    };
  } catch (err) {
    return { ok: false, error: '通信エラー: ' + err };
  }
}

function dailyAggregate(dateStrOpt) {
  var dateStr = dateStrOpt || addDaysStr_(todayStr_(), -1);
  var agg = computeDailyRanking_(dateStr);

  var rankSheet = sheet_(SHEETS.DAILY_RANK);
  var rankData = readSheetAsObjects_(rankSheet);
  deleteRowsWhere_(rankSheet, rankData.rows, function (r) { return normDateStr_(r.date) === dateStr; });

  var rankObjs = agg.students.map(function (e) {
    return {
      date: dateStr, student_id: e.student_id, display_name: e.display_name, class: e.class, course: e.course,
      role: e.role, score: e.score, elapsed_sec: e.elapsed_sec,
      rank_class: e.rank_class || '', n_class: e.n_class || '',
      rank_course: e.rank_course || '', n_course: e.n_course || '',
      rank_grade: e.rank_grade, n_grade: e.n_grade
    };
  });
  appendObjectRows_(rankSheet, HEADERS[SHEETS.DAILY_RANK], rankObjs);

  var classSheet = sheet_(SHEETS.DAILY_CLASS);
  var classData = readSheetAsObjects_(classSheet);
  deleteRowsWhere_(classSheet, classData.rows, function (r) { return normDateStr_(r.date) === dateStr; });

  var classObjs = agg.classAgg.map(function (c) {
    return { date: dateStr, unit_type: 'class', unit: c.unit, avg_score: c.avg_score, join_rate: c.join_rate, n_participants: c.n_participants, n_total: c.n_total };
  }).concat(agg.courseAgg.map(function (c) {
    return { date: dateStr, unit_type: 'course', unit: c.unit, avg_score: c.avg_score, join_rate: c.join_rate, n_participants: c.n_participants, n_total: c.n_total };
  }));
  appendObjectRows_(classSheet, HEADERS[SHEETS.DAILY_CLASS], classObjs);

  // P1: stats_question を全期間の answers から再計算し全面書き直し（冪等）
  rebuildStatsQuestion_();

  logEvent_('dailyAggregate', 'trigger', dateStr + ': ' + agg.students.length + '名集計');
  return { ok: true, date: dateStr, count: agg.students.length };
}

/** answersシートの全期間を question_id で集計し、stats_questionを全面書き直し（ヘッダー保持・冪等） */
function rebuildStatsQuestion_() {
  var answers = dedupeAnswers_(readSheetAsObjects_(sheet_(SHEETS.ANSWERS)).rows);
  var stat = {};
  var order = [];
  answers.forEach(function (a) {
    var qid = String(a.question_id).trim();
    if (!qid) return;
    if (!(qid in stat)) { stat[qid] = { attempts: 0, correct: 0 }; order.push(qid); }
    stat[qid].attempts++;
    if (isTrue_(a.is_correct)) stat[qid].correct++;
  });

  var objs = order.map(function (qid) {
    var s = stat[qid];
    return {
      question_id: qid,
      attempts: s.attempts,
      correct: s.correct,
      correct_rate: s.attempts ? (s.correct / s.attempts) : 0
    };
  });

  var sheet = sheet_(SHEETS.STATS_QUESTION);
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).clearContent(); // ヘッダー保持で本体クリア
  appendObjectRows_(sheet, HEADERS[SHEETS.STATS_QUESTION], objs);
  return { ok: true, questions: objs.length };
}

function computeWeeklyRanking_(weekStart) {
  var weekEnd = addDaysStr_(weekStart, 6);
  var roster = getRosterMap_();
  var courseMap = getCourseMap_();
  var subs = dedupeSubmissions_(readSheetAsObjects_(sheet_(SHEETS.SUBMISSIONS)).rows).filter(function (s) {
    var d = normDateStr_(s.target_date);
    return d >= weekStart && d <= weekEnd;
  });

  var byStudent = {};
  subs.forEach(function (s) {
    var sid = String(s.student_id).trim();
    byStudent[sid] = byStudent[sid] || { student_id: sid, total_score: 0, total_elapsed_sec: 0, days: 0 };
    byStudent[sid].days++; // 参加日数はさかのぼり含む
    if (!isTrue_(s.is_makeup)) {
      byStudent[sid].total_score += Number(s.total_score) || 0;
      byStudent[sid].total_elapsed_sec += Number(s.total_elapsed_sec) || 0;
    }
  });

  var students = [], teachers = [];
  Object.keys(byStudent).forEach(function (sid) {
    var r = roster[sid];
    if (!r) return;
    var e = byStudent[sid];
    e.display_name = displayNameOf_(r);
    e.rank_name = rankingNameOf_(r);
    e.class = r.class;
    e.course = resolveCourse_(r, courseMap);
    e.role = r.role;
    if (r.role === 'teacher') teachers.push(e); else students.push(e);
  });

  function sortFn(a, b) {
    if (b.total_score !== a.total_score) return b.total_score - a.total_score;
    return a.total_elapsed_sec - b.total_elapsed_sec;
  }
  students.sort(sortFn);
  students.forEach(function (e, i) { e.rank_grade = i + 1; });
  ['class', 'course'].forEach(function (dim) {
    var groups = {};
    students.forEach(function (e) { if (e[dim]) { groups[e[dim]] = groups[e[dim]] || []; groups[e[dim]].push(e); } });
    Object.keys(groups).forEach(function (key) {
      groups[key].sort(sortFn).forEach(function (e, i) { e['rank_' + dim] = i + 1; });
    });
  });

  return { weekStart: weekStart, weekEnd: weekEnd, students: students, teachers: teachers };
}

function weeklyAggregate(anyDateInWeekOpt) {
  var base = anyDateInWeekOpt || addDaysStr_(todayStr_(), -1);
  var weekStart = mondayOfWeek_(base);
  var agg = computeWeeklyRanking_(weekStart);

  var sheet = sheet_(SHEETS.WEEKLY_RANK);
  var data = readSheetAsObjects_(sheet);
  deleteRowsWhere_(sheet, data.rows, function (r) { return normDateStr_(r.week_start) === weekStart; });

  var objs = agg.students.map(function (e) {
    return {
      week_start: weekStart, student_id: e.student_id, display_name: e.display_name, class: e.class, course: e.course,
      role: e.role, total_score: e.total_score, total_elapsed_sec: e.total_elapsed_sec, days_participated: e.days,
      rank_class: e.rank_class || '', rank_course: e.rank_course || '', rank_grade: e.rank_grade
    };
  });
  appendObjectRows_(sheet, HEADERS[SHEETS.WEEKLY_RANK], objs);

  logEvent_('weeklyAggregate', 'trigger', weekStart + '週: ' + agg.students.length + '名集計');
  return { ok: true, weekStart: weekStart, count: agg.students.length };
}

/** ランキング画面「ウィークリー」タブ用（常に最新状態をライブ計算） */
function getWeeklyBoard(anyDateInWeekOpt) {
  try {
    var base = normDateStr_(anyDateInWeekOpt) || addDaysStr_(todayStr_(), -1);
    var weekStart = mondayOfWeek_(base);
    var agg = computeWeeklyRanking_(weekStart);
    var top20 = topWithTies_(agg.students, 20, 'total_score').map(function (e) {
      return { rank: e.rank_grade, name: e.rank_name, score: e.total_score, days: e.days, student_id: e.student_id };
    });
    var teacherRefs = agg.teachers.map(function (t) {
      return { name: t.rank_name, score: t.total_score, days: t.days };
    });
    return { ok: true, weekStart: weekStart, weekEnd: agg.weekEnd, top20: top20, teacherRefs: teacherRefs };
  } catch (err) {
    return { ok: false, error: '通信エラー: ' + err };
  }
}

function monthlyAggregate(monthOpt) {
  var month = monthOpt || monthOf_(addDaysStr_(todayStr_(), -1));

  var roster = getRosterMap_();
  var courseMap = getCourseMap_();
  var subs = dedupeSubmissions_(readSheetAsObjects_(sheet_(SHEETS.SUBMISSIONS)).rows).filter(function (s) {
    return normDateStr_(s.target_date).indexOf(month) === 0;
  });
  var answers = dedupeAnswers_(readSheetAsObjects_(sheet_(SHEETS.ANSWERS)).rows).filter(function (a) {
    return normDateStr_(a.target_date).indexOf(month) === 0;
  });
  var questionsMap = getQuestionsMap_();

  var byStudent = {};
  subs.forEach(function (s) {
    var sid = String(s.student_id).trim();
    byStudent[sid] = byStudent[sid] || { student_id: sid, total_score: 0, days: 0 };
    byStudent[sid].days++;
    if (!isTrue_(s.is_makeup)) byStudent[sid].total_score += Number(s.total_score) || 0;
  });

  var genreByStudent = {};
  answers.forEach(function (a) {
    var sid = String(a.student_id).trim();
    var q = questionsMap[String(a.question_id).trim()];
    if (!q) return;
    genreByStudent[sid] = genreByStudent[sid] || {};
    var g = q.genre || '未分類';
    genreByStudent[sid][g] = genreByStudent[sid][g] || { ok: 0, all: 0 };
    genreByStudent[sid][g].all++;
    if (isTrue_(a.is_correct)) genreByStudent[sid][g].ok++;
  });

  var publishDaysThisMonth = {};
  Object.keys(questionsMap).forEach(function (id) {
    var pd = questionsMap[id].publish_date;
    if (pd.indexOf(month) === 0) publishDaysThisMonth[pd] = true;
  });
  var totalSchoolDays = Object.keys(publishDaysThisMonth).length;

  var students = [], teachers = [];
  Object.keys(byStudent).forEach(function (sid) {
    var r = roster[sid];
    if (!r) return;
    var e = byStudent[sid];
    e.display_name = displayNameOf_(r);
    e.class = r.class;
    e.course = resolveCourse_(r, courseMap);
    e.role = r.role;
    var best = '', bestRate = -1;
    var gs = genreByStudent[sid] || {};
    Object.keys(gs).forEach(function (g) {
      var rate = gs[g].ok / gs[g].all;
      if (gs[g].ok > 0 && rate > bestRate) { bestRate = rate; best = g; }
    });
    e.best_genre = best;
    e.is_kaikin = totalSchoolDays > 0 && e.days >= totalSchoolDays;
    if (r.role === 'teacher') teachers.push(e); else students.push(e);
  });

  students.sort(function (a, b) { return b.total_score - a.total_score; });
  students.forEach(function (e, i) { e.rank_grade = i + 1; });

  var sheet = sheet_(SHEETS.MONTHLY_RANK);
  var data = readSheetAsObjects_(sheet);
  deleteRowsWhere_(sheet, data.rows, function (r) { return String(r.month).trim() === month; });

  var objs = students.map(function (e) {
    return {
      month: month, student_id: e.student_id, display_name: e.display_name, class: e.class, course: e.course,
      role: e.role, total_score: e.total_score, days_participated: e.days, rank_grade: e.rank_grade,
      best_genre: e.best_genre, is_kaikin: e.is_kaikin
    };
  });
  appendObjectRows_(sheet, HEADERS[SHEETS.MONTHLY_RANK], objs);

  logEvent_('monthlyAggregate', 'trigger', month + ': ' + students.length + '名集計');
  return { ok: true, month: month, count: students.length };
}

function archiveLogs(monthOpt) {
  var month = monthOpt || monthOf_(addDaysStr_(todayStr_(), -1));
  var year = month.slice(0, 4);
  var archiveName = 'quiz_answers_archive_' + year;

  var files = DriveApp.getFilesByName(archiveName);
  var archiveSs;
  if (files.hasNext()) {
    archiveSs = SpreadsheetApp.open(files.next());
  } else {
    archiveSs = SpreadsheetApp.create(archiveName);
  }

  var archiveSheet = archiveSs.getSheetByName(month);
  if (!archiveSheet) {
    archiveSheet = archiveSs.insertSheet(month);
    archiveSheet.getRange(1, 1, 1, HEADERS[SHEETS.ANSWERS].length).setValues([HEADERS[SHEETS.ANSWERS]]);
  }

  var answersSheet = sheet_(SHEETS.ANSWERS);
  var data = readSheetAsObjects_(answersSheet);
  var targetRows = data.rows.filter(function (r) { return normDateStr_(r.target_date).indexOf(month) === 0; });

  if (targetRows.length) {
    appendObjectRows_(archiveSheet, HEADERS[SHEETS.ANSWERS], targetRows);
    deleteRowsWhere_(answersSheet, targetRows, function () { return true; });
  }

  logEvent_('archiveLogs', 'trigger', month + ': ' + targetRows.length + '行を' + archiveName + 'へ退避');
  return { ok: true, month: month, movedRows: targetRows.length, archiveUrl: archiveSs.getUrl() };
}

function setupSpreadsheet() {
  Object.keys(SHEETS).forEach(function (key) { ensureSheet_(SHEETS[key]); });

  var settingsSheet = sheet_(SHEETS.SETTINGS);
  var existing = readSheetAsObjects_(settingsSheet).rows.map(function (r) { return String(r.key).trim(); });
  DEFAULT_SETTINGS.forEach(function (kv) {
    if (existing.indexOf(kv[0]) === -1) {
      appendObjectRow_(settingsSheet, HEADERS[SHEETS.SETTINGS], { key: kv[0], value: kv[1] });
    }
  });

  // course_map のひな型（class 1〜8, course空欄）。既存があれば触らない
  var cmSheet = sheet_(SHEETS.COURSE_MAP);
  var cmExisting = readSheetAsObjects_(cmSheet).rows.map(function (r) { return String(r.class).trim(); });
  for (var c = 1; c <= 8; c++) {
    var key = String(c);
    if (cmExisting.indexOf(key) === -1) {
      appendObjectRow_(cmSheet, HEADERS[SHEETS.COURSE_MAP], { class: key, course: '' });
    }
  }

  // 日付系の列を書式「書式なしテキスト」にして自動で日付型変換されるのを防ぐ
  [
    [SHEETS.QUESTIONS, ['publish_date']],
    [SHEETS.ANSWERS, ['target_date']],
    [SHEETS.SUBMISSIONS, ['target_date']],
    [SHEETS.DAILY_RANK, ['date']],
    [SHEETS.DAILY_CLASS, ['date']],
    [SHEETS.WEEKLY_RANK, ['week_start']],
    [SHEETS.MONTHLY_RANK, ['month']],
    [SHEETS.SETTINGS, ['value']]
  ].forEach(function (pair) {
    var sheet = sheet_(pair[0]);
    pair[1].forEach(function (colName) {
      var colIdx = HEADERS[pair[0]].indexOf(colName) + 1;
      if (colIdx > 0) sheet.getRange(1, colIdx, Math.max(sheet.getMaxRows() - 1, 1000), 1).setNumberFormat('@');
    });
  });

  logEvent_('setupSpreadsheet', 'menu', 'シート初期化完了');
  return { ok: true };
}

function importRoster() {
  var srcSs = SpreadsheetApp.openById(ROSTER_SOURCE_SS_ID);
  var srcSheet = srcSs.getSheetByName(ROSTER_SOURCE_SHEET);
  if (!srcSheet) throw new Error('原本シート「' + ROSTER_SOURCE_SHEET + '」が見つかりません');

  var values = srcSheet.getDataRange().getValues();
  var rosterSheet = sheet_(SHEETS.ROSTER);
  var existing = readSheetAsObjects_(rosterSheet);
  var existingMap = {};
  existing.rows.forEach(function (r) { existingMap[String(r.student_id).trim()] = r; });

  var added = 0, updated = 0;
  // ヘッダー行がある場合は1行目を飛ばす（A列が「生徒ID」等の文字列でIDパターンに一致しない場合）
  var headerLikely = values.length && !studentIdFormatValid_(String(values[0][0]).trim());
  var dataStart = headerLikely ? 1 : 0;

  var toAppend = [];
  for (var i = dataStart; i < values.length; i++) {
    var id = String(values[i][0] || '').trim();
    var name = String(values[i][1] || '').trim();
    if (!id) continue;
    var cls = classFromStudentId_(id);
    if (existingMap[id]) {
      // 既存の nickname / course / homeroom_class / active は保持。name のみ更新反映
      var rowIdx = existingMap[id].__row;
      if (String(existingMap[id].name) !== name) {
        rosterSheet.getRange(rowIdx, HEADERS[SHEETS.ROSTER].indexOf('name') + 1).setValue(name);
        updated++;
      }
    } else {
      toAppend.push({
        student_id: id, name: name, nickname: '', class: cls, course: '', role: 'student', homeroom_class: '', active: true
      });
      added++;
    }
  }
  appendObjectRows_(rosterSheet, HEADERS[SHEETS.ROSTER], toAppend);

  invalidateRosterCache_(); // 名簿更新を即反映
  var message = added + '件追加 / ' + updated + '件更新（course・nicknameは保持）';
  logEvent_('importRoster', 'menu', message);
  return { ok: true, added: added, updated: updated, message: message };
}

function importQuestionsCsv() {
  var questionsSheet = sheet_(SHEETS.QUESTIONS);
  var existingIds = {};
  readSheetAsObjects_(questionsSheet).rows.forEach(function (r) {
    existingIds[String(r.question_id).trim()] = true;
  });

  var totalAdded = 0, totalSkipped = 0, filesFound = 0;
  var messages = [];

  // Drive全体から questions_*.csv を検索し、ファイル名順に取込む（同名重複は最終更新が新しい方を採用）
  var candidates = {};
  var it = DriveApp.searchFiles('title contains "questions_"');
  while (it.hasNext()) {
    var f = it.next();
    var name = f.getName();
    if (!QUESTION_CSV_PATTERN.test(name)) continue;
    if (!candidates[name] || f.getLastUpdated() > candidates[name].getLastUpdated()) {
      candidates[name] = f;
    }
  }
  var names = Object.keys(candidates).sort();
  if (!names.length) {
    var noneMsg = 'Driveに questions_*.csv が見つかりません（アップロード後に再実行してください）';
    logEvent_('importQuestionsCsv', 'menu', noneMsg);
    return { ok: false, added: 0, skipped: 0, message: noneMsg };
  }

  names.forEach(function (filename) {
    filesFound++;
    var file = candidates[filename];
    var text = file.getBlob().getDataAsString('UTF-8');
    var table = Utilities.parseCsv(text);
    if (!table.length) { messages.push(filename + ': 空ファイル'); return; }

    var header = table[0].map(function (h) { return String(h).trim(); });
    var toAppend = [];
    var added = 0, skipped = 0, invalid = 0;

    for (var i = 1; i < table.length; i++) {
      var rowArr = table[i];
      if (!rowArr.length || rowArr.every(function (c) { return c === ''; })) continue;
      var obj = {};
      header.forEach(function (h, idx) { obj[h] = rowArr[idx]; });

      var qid = String(obj.question_id || '').trim();
      if (!qid) continue;
      if (existingIds[qid]) { skipped++; continue; }

      var answerNum = Number(obj.answer);
      if (!(answerNum >= 1 && answerNum <= 4)) { invalid++; continue; }

      ['passage', 'question_text', 'choice1', 'choice2', 'choice3', 'choice4', 'explanation'].forEach(function (f) {
        obj[f] = String(obj[f] || '').replace(/\\n/g, '\n');
      });
      obj.seq = Number(obj.seq);
      obj.difficulty = Number(obj.difficulty);
      obj.time_limit_sec = Number(obj.time_limit_sec) || 60;
      obj.publish_date = normDateStr_(obj.publish_date);

      toAppend.push(obj);
      existingIds[qid] = true;
      added++;
    }
    appendObjectRows_(questionsSheet, HEADERS[SHEETS.QUESTIONS], toAppend);
    totalAdded += added; totalSkipped += skipped;
    messages.push(filename + ': ' + added + '件追加 / ' + skipped + '件重複スキップ / ' + invalid + '件不正');
  });

  invalidateQuestionsCache_(); // 問題日付スライスキャッシュを全消去し即反映
  var message = filesFound + 'ファイル処理: 計' + totalAdded + '件追加, ' + totalSkipped + '件重複スキップ\n' + messages.join('\n');
  logEvent_('importQuestionsCsv', 'menu', message);
  return { ok: true, added: totalAdded, skipped: totalSkipped, message: message };
}

/* P8: 当日クイズ先読み。「本日」該当かつ未回答の日があれば裏でgetQuizを先行発火 */
function prefetchTodayQuiz(){
  prefetchQuiz = null;
  var todays = availableDates.filter(function(d){
    return d.hasQuestions && !d.done && !d.isMakeup && d.date === todayStr;
  });
  if(!todays.length) return;
  var date = todays[0].date;
  var state = { date: date, status: "loading", data: null };
  prefetchQuiz = state;
  run("getQuiz", [me.id, date], function(res){
    if(prefetchQuiz !== state) return; // 破棄済み（再描画等）なら無視
    state.status = "done";
    state.data = res;
  }, function(){
    if(prefetchQuiz !== state) return;
    state.status = "error";
  });
}

/* ================= 出題取得 ================= */
function startQuiz(dateStr){
  // P8: プリフェッチ済み・同一dateなら通信スキップで即遷移。一度使ったら破棄（二重利用防止）
  if(prefetchQuiz && prefetchQuiz.date === dateStr && prefetchQuiz.status === "done"){
    var pf = prefetchQuiz;
    prefetchQuiz = null;
    applyQuizResult(pf.data, dateStr);
    return;
  }
  if(prefetchQuiz && prefetchQuiz.date === dateStr && prefetchQuiz.status === "loading"){
    // まだ先読み中: 新規通信は発火せず、現行どおりローディング表示のまま先読み完了を待つ
    app.innerHTML = '<div class="card loading">問題を読み込み中…</div>';
    waitPrefetch(prefetchQuiz, dateStr);
    return;
  }
  if(prefetchQuiz && prefetchQuiz.date === dateStr && prefetchQuiz.status === "error"){
    prefetchQuiz = null; // 失敗時は現行フローにフォールバック
  }
  app.innerHTML = '<div class="card loading">問題を読み込み中…</div>';
  run("getQuiz", [me.id, dateStr], function(res){
    applyQuizResult(res, dateStr);
  }, function(err){ alert("通信エラー: " + err); renderDaySelect(); });
}

/* P8: 先読み中の場合、完了 or 失敗をポーリングして待つ（新規通信は発火しない） */
function waitPrefetch(state, dateStr){
  if(prefetchQuiz !== state){
    // 別の遷移等で破棄済み。フォールバックとして通常取得
    app.innerHTML = '<div class="card loading">問題を読み込み中…</div>';
    run("getQuiz", [me.id, dateStr], function(res){ applyQuizResult(res, dateStr); },
      function(err){ alert("通信エラー: " + err); renderDaySelect(); });
    return;
  }
  if(state.status === "loading"){
    setTimeout(function(){ waitPrefetch(state, dateStr); }, 150);
    return;
  }
  if(state.status === "done"){
    prefetchQuiz = null;
    applyQuizResult(state.data, dateStr);
    return;
  }
  // error: フォールバックで通常取得
  prefetchQuiz = null;
  run("getQuiz", [me.id, dateStr], function(res){ applyQuizResult(res, dateStr); },
    function(err){ alert("通信エラー: " + err); renderDaySelect(); });
}

function applyQuizResult(res, dateStr){
  if(!res || !res.ok){ alert((res && res.error) || "通信エラー"); renderDaySelect(); return; }
  quizQuestions = res.questions;
  quizTargetDate = res.targetDate;
  quizIsMakeup = res.isMakeup;
  idx = 0;
  renderQ();
}

function renderQ(){
  var q = quizQuestions[idx];
  timeLeft = q.time_limit_sec || 60;
  qStartMs = Date.now();
  var stars = "★".repeat(q.difficulty) + "☆".repeat(3-q.difficulty);

     '<div class="bar"><div id="tbar" style="width:100%"></div></div>' +
     '<div class="qa">' +
       '<div class="qleft">' +
         '<span class="tag">' + q.genre + '</span><span class="stars">' + stars + '（' + ptMap[q.difficulty] + '点）</span>' +
         (q.passage ? '<div class="passage">' + passageToHtml(q.passage) + '</div>' : "") +
         '<div class="qtext">' + escapeHtml(q.question_text).replace(/\n/g,"<br>") + '</div>' +
       '</div>' +
       '<div class="qright">' +
         '<div id="choices">' + q.choices.map(function(c,i){
           return '<button class="choice" data-i="' + i + '">' + ["①","②","③","④"][i] + " " + escapeHtml(c) + '</button>';
         }).join("") + '</div>' +
         '<div id="fb" class="fb"></div>' +
       '</div>' +
     '</div>' +
   '</div>';
  Array.prototype.forEach.call(document.querySelectorAll(".choice"), function(b){
    b.onclick = function(){ answer(Number(b.getAttribute("data-i"))); };
  });
  var total = timeLeft;
  timer = setInterval(function(){
    timeLeft--;
    document.getElementById("tl").textContent = timeLeft + "秒";
    var bar = document.getElementById("tbar");
    bar.style.width = (timeLeft/total*100) + "%";
    if(timeLeft <= 10) bar.classList.add("warn");
    if(timeLeft <= 0){ clearInterval(timer); answer(-1); }
  }, 1000);
}

function answer(dispIdx){
  clearInterval(timer);
  var q = quizQuestions[idx];
  var elapsed = Math.round((Date.now() - qStartMs)/1000);
  var origIdx1based = dispIdx >= 0 ? (q.dispToOrig[dispIdx] + 1) : 0;

  Array.prototype.forEach.call(document.querySelectorAll(".choice"), function(b){ b.disabled = true; });
  var fb = document.getElementById("fb");
  fb.className = "fb wait";
  fb.innerHTML = "採点中…";

  run("submitAnswer", [me.id, quizTargetDate, q.question_id, origIdx1based, elapsed], function(res){
    if(!res.ok){ fb.className="fb ng"; fb.innerHTML = "エラー: " + res.error +
      '<button class="btn" id="btnNext">次へ</button>'; document.getElementById("btnNext").onclick=next; return; }
    var ok = res.correct;
    Array.prototype.forEach.call(document.querySelectorAll(".choice"), function(b,i){
      var orig = q.dispToOrig[i] + 1;
      if(orig === res.answer) b.classList.add(i===dispIdx ? "sel-ok" : "reveal");
      else if(i === dispIdx) b.classList.add("sel-ng");
    });
    fb.className = "fb " + (ok ? "ok" : "ng");
    fb.innerHTML = (ok ? "⭕ 正解！ +" + res.point + "点" : (dispIdx<0 ? "⏰ 時間切れ…" : "❌ ざんねん…")) +
      "<br>【解説】" + escapeHtml(res.explanation).replace(/\n/g,"<br>") +
      '<button class="btn" id="btnNext">' + (idx+1<quizQuestions.length ? "次の問題へ" : "結果を見る") + '</button>';
    document.getElementById("btnNext").onclick = next;
  });
}
function next(){
  idx++;
  if(idx < quizQuestions.length) renderQ(); else finishAndShowResult();
}

/* ================= 画面: 結果 ================= */
function finishAndShowResult(){
  app.innerHTML = '<div class="card loading">採点中…</div>';
  run("finishQuiz", [me.id, quizTargetDate], function(res){
    if(!res.ok){ alert(res.error); renderDaySelect(); return; }
    if(!settings.demo_mode){
      availableDates.forEach(function(d){ if(d.date === quizTargetDate) d.done = true; });
    }
    lastResult = res;
    if(res.isMakeup){
      renderResult();
    } else if(res.results){
      // P7: finishQuizの戻りにgetResults相当が同梱済みなら1往復で完結
      lastResultsData = res.results.ok ? res.results : null;
      renderResult();
    } else {
      // フォールバック: results未同梱時（想定外ケース）は従来どおり別途取得
      run("getResults", [me.id, quizTargetDate], function(rres){
        lastResultsData = rres.ok ? rres : null;
        renderResult();
      }, function(){ lastResultsData = null; renderResult(); });
    }
  });
}

function renderResult(){
  var gs = lastResult.genreStats || {};
  var okCnt = 0, allCnt = 0;
  Object.keys(gs).forEach(function(g){ okCnt += gs[g].ok; allCnt += gs[g].all; });
  var best = "－", bestRate = -1;
  Object.keys(gs).forEach(function(g){
    var r = gs[g].ok/gs[g].all;
    if(gs[g].ok>0 && r>bestRate){ bestRate=r; best=g; }
  });
  var maxScore = 0;
  quizQuestions.forEach(function(q){ maxScore += ({1:settings.pt_1,2:settings.pt_2,3:settings.pt_3})[q.difficulty] || 0; });
  maxScore += settings.join_bonus;

  var ranksHtml = "";
  if(!lastResult.isMakeup && lastResultsData && lastResultsData.myRanks){
    var mr = lastResultsData.myRanks;
    ranksHtml =
     '<div class="card">' +
       '<h2>あなたの順位（本日集計）</h2>' +
       '<div class="myrank">' +

function drawRadar(genreStat){
  var cv = document.getElementById("radar"); if(!cv) return;
  var ctx = cv.getContext("2d");
  var css = getComputedStyle(document.documentElement);
  var colBorder = css.getPropertyValue("--border").trim();
  var colSub = css.getPropertyValue("--sub").trim();
  var colFill = css.getPropertyValue("--chart-fill").trim();
  var colAcc = css.getPropertyValue("--acc").trim();
  var genres = Object.keys(genreStat).length ? Object.keys(genreStat) : ["推論","数理","ことば","文脈","読解","資料","英語","教養"];
  var cx=150, cy=125, R=85, N=genres.length;
  function pt(i,r){ var a=-Math.PI/2+i*2*Math.PI/N; return [cx+r*Math.cos(a), cy+r*Math.sin(a)]; }
  ctx.strokeStyle=colBorder; ctx.fillStyle=colSub; ctx.font="12px sans-serif";
  for(var ring=1; ring<=4; ring++){
    ctx.beginPath();
    for(var i=0;i<=N;i++){ var p=pt(i%N, R*ring/4); i?ctx.lineTo(p[0],p[1]):ctx.moveTo(p[0],p[1]); }
    ctx.stroke();
  }
  genres.forEach(function(g,i){
    var p=pt(i,R+18);
    ctx.textAlign="center"; ctx.fillText(g,p[0],p[1]+4);
  });
  ctx.beginPath();
  genres.forEach(function(g,i){
    var s=genreStat[g], r=s? (s.ok/s.all)*R : 0;
    var p=pt(i, Math.max(r,4));
    i?ctx.lineTo(p[0],p[1]):ctx.moveTo(p[0],p[1]);
  });
  ctx.closePath();
  ctx.fillStyle=colFill; ctx.fill();
  ctx.strokeStyle=colAcc; ctx.lineWidth=2; ctx.stroke();
}

/* ================= util ================= */
function escapeHtml(s){
  return String(s||"").replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;");
}
function esc(s){ return escapeHtml(s); }

/** passage文字列をHTMLへ変換する。
 *  行頭が「|」の連続行は表として描画（1行目をヘッダー扱い）。それ以外の行は<br>区切りのテキスト。
 *  例: |クラス|人数|\n|1組|32| → 2列の表 */
function passageToHtml(p){
  var lines = String(p||"").split("\n");
  var html = "", tableRows = [], textBuf = [];
  function flushText(){
    if(textBuf.length){ html += textBuf.map(escapeHtml).join("<br>"); textBuf = []; }
  }
  function flushTable(){
    if(!tableRows.length) return;
    var t = '<table class="ptable">';
    tableRows.forEach(function(cells, ri){
      var tag = (ri === 0) ? "th" : "td";
      t += "<tr>" + cells.map(function(c){ return "<" + tag + ">" + escapeHtml(c) + "</" + tag + ">"; }).join("") + "</tr>";
    });
    t += "</table>";
    html += t;
    tableRows = [];
  }
  lines.forEach(function(line){
    if(/^\s*\|/.test(line)){
      flushText();
      var cells = line.replace(/^\s*\|/, "").replace(/\|\s*$/, "").split("|").map(function(c){ return c.trim(); });
      tableRows.push(cells);
    }else{
      flushTable();
      textBuf.push(line);
    }
  });
  flushText(); flushTable();
  return html;
}
